	float altura = 0;

	}
	while (altura <= 0) {
		cout << "Ingrese la altura \n";
		cin >> altura;
	}
	float IMC;
	IMC = peso / (altura*altura);
	if (IMC < 18.5) {
		cout << "La persona esta con bajo peso\n";
	}

	else if (IMC >= 18.5 and IMC < 25.0) {
		cout << "Tiene un peso normal\n";
	}
	else if (IMC >= 25.0 and IMC < 30) {
		cout << "Tiene sobrepeso\n ";
	}
	else if (IMC >= 30) {
		cout << "Tiene Obesidad\n";
			
	}
